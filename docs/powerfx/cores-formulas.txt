## Status — regra de produto

# Em andamento / Em campo  = azul  (Primary)
# Concluido / Finalizada   = verde (Success)
# Pendente                 = cinza (Neutral)
#
# NAO usar o Switch antigo da Tela_SelecionarInventario (estava invertido).
# Chip das campanhas no Mobile: Tela_SelecionarInventario → Gallery2 → Button3_13
# (Fill e Color). Pendente cai no default Neutral — nao usar Cor.Warning.

## Chip Fill

Switch(
    ThisItem.Status;
    "Em andamento"; Cor.PrimaryL90;
    "Em campo"; Cor.PrimaryL90;
    "Concluído"; Cor.SuccessL70;
    "Finalizada"; Cor.SuccessL70;
    Cor.NeutralL90
)

## Chip Color (texto)

Switch(
    ThisItem.Status;
    "Em andamento"; Cor.PrimaryD30;
    "Em campo"; Cor.PrimaryD30;
    "Concluído"; Cor.SuccessD30;
    "Finalizada"; Cor.SuccessD30;
    Cor.NeutralL30
)
