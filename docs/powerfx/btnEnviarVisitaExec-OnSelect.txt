                        Observacoes: linhaAtual.Observacoes;

                                linhaAtual.CoordX;
                                LookUp(
                                    colInventarioBase;

                                Full: fotoJson;
                                Large: fotoJson;
                                Medium: fotoJson;
                                Small: fotoJson
                            }
                        }

                );

                    colFotosCampo;

                )

        Filter(
            colSubestacoes;
            CodInventario = gblInventarioSelecionado.CodInventario
            && Status = "Finalizada"

                    Coalesce(seFin.ID; 0) > 0;

                    LookUp(
                        Subestacoes;

            colCampanhas;
            CodInventario = cod;
            {Status: "Em andamento"}
