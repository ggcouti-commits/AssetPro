        // —— Revalidar sessão (SessaoVersao) só aqui ——

                        {__: true};
                        Set(gblUsuario; Blank());;
                        Set(varPerfil; Blank());;

                    )
                )
            )
        );;

        ClearCollect(colUltimaSinc; {DataHora: Now()});;

        Set(gblUltimaSincronizacao; Now());;

    )
