### 0b. Tela_Exportar → OnVisible

# Volta na tela: so atualiza a barra do card. Nao reabre o popup.

```
If(
    varOpArquivo = "IMPORTANDO" && !IsBlank(varImportResumo.Codigo);
    With(
        {p: flxProgressoImportacao.Run(varImportResumo.Codigo)};
        Set(varImportSEsCriadas; Value(p.ses));;
        Set(varImportCriados; Value(p.criados));;
        If(
            Value(p.ses) >= varImportTotalSEs && Value(p.criados) >= varImportTotal;
            Set(varOpArquivo; "");;
            Set(varBarraCard; false);;
            Set(varLockCod; "");;
            Patch(
                'Campanhas ';
                LookUp('Campanhas '; CodInventario = varImportResumo.Codigo);
                {Status: "Pendente"}
            );;
            Patch(
                colCampanhas;
                LookUp(colCampanhas; CodInventario = varImportResumo.Codigo);
                {Status: "Pendente"}
            );;
            ClearCollect(
                colSubestacoes;
                Filter(
                    Subestacoes;
                    CodInventario = varImportResumo.Codigo
                )
            )
        )
    );
    varOpArquivo = "EXCLUINDO" && !IsBlank(varLockCod);
    With(
        {p: flxProgressoImportacao.Run(varLockCod)};
        Set(varImportSEsCriadas; Value(p.ses));;
        Set(varImportCriados; Value(p.criados));;
        If(
            varExcluirTotal = 0 && Value(p.criados) > 0;
            Set(varExcluirTotal; Value(p.criados))
        );;
        If(
            varExcluirTotalSEs = 0 && Value(p.ses) > 0;
            Set(varExcluirTotalSEs; Value(p.ses))
        );;
        If(
            Value(p.ses) = 0 && Value(p.criados) = 0;
            Set(varOpArquivo; "");;
            Set(varBarraCard; false);;
            Set(varLockCod; "");;
            Set(varCampExcluir; Blank());;
            Refresh('Campanhas ');;
            Refresh(Subestacoes);;
            Clear(colSubestacoes);;
            ForAll(
                Filter('Campanhas '; Ativo = "Sim") As camp;
                Collect(
                    colSubestacoes;
                    Filter(Subestacoes; CodInventario = camp.CodInventario)
                )
            );;
            ClearCollect(
                colCampanhas;
                AddColumns(
                    Filter('Campanhas '; Ativo = "Sim");
                    QtdSEs;
                    With(
                        {cod: ThisRecord.CodInventario};
                        CountRows(Filter(colSubestacoes; CodInventario = cod))
                    );
                    QtdItens;
                    0
                )
            )
        )
    )
)
```

### 20p. Barra de progresso — processo (decisao 13.09: "quase total")

# O Responder_OK subiu no fluxo: agora e a primeira acao do ramo `Se nao` do JaImportou,
# antes do Listar_SEs (ver docs/automate/flxImportarCampanha.md, passo 18). O App e solto
# em ~12 s e a barra cobre SEs + pastas + itens — 89% do tempo, contra 63% de antes.
# Fica cego so upload + script + pasta raiz, porque durante o Run o App esta bloqueado:
# Timer nao dispara, Set nao roda. Ali e spinner (varCarregando), nao tem volta.

# Peso: 1 SE = 3 chamadas ao SharePoint (item + pasta da SE + pasta Visão geral),
# 1 item = 1 chamada. Amostra 5 SEs / 65 itens => 15 + 65 = 80 unidades.
# Dividir em "50% SEs / 50% itens" mentiria: a barra ficaria parada em 50% no minuto
# mais longo e o usuario acha que travou.

#### Fluxo `flxProgressoImportacao` (novo, leve)

# 1. Gatilho Power Apps (V2) → uma entrada Texto: CodInventario
# 2. SharePoint — Obter itens — `Obter_Criados`
#    Lista: Inventario
#    Consulta de filtro: CodInventario eq '@{triggerBody()?['text']}'
#    Selecionar consulta: ID          (so o ID: resposta pequena, chamada rapida)
#    Contagem superior: 5000 · Paginacao ligada 5000
# 2b. SharePoint — Obter itens — `Obter_SEs_criadas`
#    Lista: Subestacoes · mesmo filtro · Selecionar consulta: ID · limite 100
# 3. Responder ao Power App — duas saidas (Número):
#    criados = length(body('Obter_Criados')?['value'])
#    ses     = length(body('Obter_SEs_criadas')?['value'])
# Publicar. No Studio: Dados → adicionar o fluxo em Power Automate.
# Duas chamadas por tick, so ID nas duas: ~300 ms. Timer de 3 s aguenta sem fila.

#### 20p-0. App.Formulas — duas formulas nomeadas

# A condicao "ainda importando" aparece em 7 controles (titulo, cor do titulo, Fill do
# icone, Image do icone, htmProgTexto, cProgTrilha.Visible, tmrProgresso.Start).
# Repetir a expressao em 7 lugares e onde nasce divergencia. Formula nomeada resolve —
# recalcula sozinha quando qualquer uma das quatro variaveis muda.

```
ImportandoAgora =
    varImportSEsCriadas < varImportTotalSEs
        || varImportCriados < varImportTotal;

ArquivoOcupado =
    varOpArquivo = "IMPORTANDO" || varOpArquivo = "EXCLUINDO";

ProgressoImport =
    Min(
        1;
        (varImportSEsCriadas * 3 + varImportCriados)
            / Max(1; varImportTotalSEs * 3 + varImportTotal)
    );

ProgressoExcluir =
    If(
        varExcluirTotalSEs + varExcluirTotal <= 0;
        0;
        Min(
            1;
            1 - (varImportSEsCriadas + varImportCriados)
                / Max(1; varExcluirTotalSEs + varExcluirTotal)
        )
    );

ProgressoCard =
    If(
        varUltimaOp = "EXCLUINDO";
        ProgressoExcluir;
        ProgressoImport
    );
```

```
        Set(varImportSEsCriadas; Value(p.ses));;
        Set(varImportCriados; Value(p.criados));;
        If(
            Value(p.ses) >= varImportTotalSEs && Value(p.criados) >= varImportTotal;
            Set(varOpArquivo; "");;
            Set(varBarraCard; false);;
            Set(varLockCod; "");;
            Patch(
                'Campanhas ';
                LookUp('Campanhas '; CodInventario = varImportResumo.Codigo);
                {Status: "Pendente"}
            );;
            Patch(
                colCampanhas;
                LookUp(colCampanhas; CodInventario = varImportResumo.Codigo);
                {Status: "Pendente"}
            );;
            ClearCollect(
                colSubestacoes;
                Filter(
                    Subestacoes;
                    CodInventario = varImportResumo.Codigo
                )
            )
        )
    );
    varOpArquivo = "EXCLUINDO";
    With(
        {p: flxProgressoImportacao.Run(varLockCod)};
        Set(varImportSEsCriadas; Value(p.ses));;
        Set(varImportCriados; Value(p.criados));;
        If(
            varExcluirTotal = 0 && Value(p.criados) > 0;
            Set(varExcluirTotal; Value(p.criados))
        );;
        If(
            varExcluirTotalSEs = 0 && Value(p.ses) > 0;
            Set(varExcluirTotalSEs; Value(p.ses))
        );;
        If(
            Value(p.ses) = 0 && Value(p.criados) = 0;
            Set(varOpArquivo; "");;
            Set(varBarraCard; false);;
            Set(varLockCod; "");;
            Set(varCampExcluir; Blank());;
            Refresh('Campanhas ');;
            Refresh(Subestacoes);;
            Clear(colSubestacoes);;
            ForAll(
                Filter('Campanhas '; Ativo = "Sim") As camp;
                Collect(
                    colSubestacoes;
                    Filter(Subestacoes; CodInventario = camp.CodInventario)
                )
            );;
            ClearCollect(
                colCampanhas;
                AddColumns(
                    Filter('Campanhas '; Ativo = "Sim");
                    QtdSEs;
                    With(
                        {cod: ThisRecord.CodInventario};
                        CountRows(Filter(colSubestacoes; CodInventario = cod))
                    );
                    QtdItens;
                    0
                )
            )
        )
    )
)
```
