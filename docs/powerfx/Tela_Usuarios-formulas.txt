## Recorte

# Slot = linha. Inventariante01 e o login; UsadoPor e quem esta com o tablet.
# Login existente e editavel. Excluir = Remove (nao so inativar).
# Ativo nesta lista e texto "Sim" / "Nao" (igual Concessionarias), nao Sim/Nao nativo.
# Toggle (varUsrAtivo) continua boolean. Na borda: le = "Sim", grava If(...; "Sim"; "Nao").
# Senha: o escritorio define aqui. Campo vazio no editar = nao troca.
# SessaoVersao (número): cada Salvar soma 1. O Mobile compara e manda de volta pro login.

### 36. btnUsrSalvar → OnSelect

Apaga o que estiver la. Cola:

```
Set(varUsrErro; Blank());;
If(
    IsBlank(Trim(inpUsrLogin.Text));
    Set(varUsrErro; "Login obrigatório.");
    varUsrNovo && IsBlank(inpUsrSenha.Text);
    Set(varUsrErro; "Senha obrigatória no slot novo.");
    !IsBlank(
        LookUp(
            Inventariantes;
            Lower(Login) = Lower(Trim(inpUsrLogin.Text))
                && (varUsrNovo || ID <> varUsrSel.ID)
        )
    );
    Set(varUsrErro; "Já existe um slot com esse login.");
    With(
        {
            seq:
                Coalesce(
                    Max(
                        AddColumns(
                            Filter(Inventariantes; StartsWith(CodUsuario; "USR-"));
                            "SeqNum";
                            IfError(Value(Mid(CodUsuario; 5; 10)); 0)
                        );
                        SeqNum
                    );
                    0
                ) + 1
        };
        Patch(
            Inventariantes;
            If(varUsrNovo; Defaults(Inventariantes); varUsrSel);
            {
                Title: Trim(inpUsrLogin.Text);
                Login: Trim(inpUsrLogin.Text);
                Perfil: varUsrPerfil;
                UsadoPor:
                    If(
                        varUsrPerfil = "INVENTARIANTE";
                        Trim(inpUsrUsadoPor.Text);
                        Blank()
                    );
                Nome: varUsrSel.Nome;
                Ativo: If(varUsrAtivo; "Sim"; "Nao");
                Senha:
                    If(
                        IsBlank(inpUsrSenha.Text);
                        varUsrSel.Senha;
                        inpUsrSenha.Text
                    );
                CodUsuario:
                    If(
                        varUsrNovo;
                        "USR-" & Text(seq; "000");
                        varUsrSel.CodUsuario
                    );
                SessaoVersao:
                    If(
                        varUsrNovo;
                        1;
                        Coalesce(varUsrSel.SessaoVersao; 0) + 1
                    )
            }
        );;
        Refresh(Inventariantes);;
        Set(varUsrSel; LookUp(Inventariantes; Login = Trim(inpUsrLogin.Text)));;
        Set(varUsrNovo; false);;
        Reset(inpUsrSenha);;
        Set(varToastMsg; "Slot salvo. O campo precisa entrar de novo.");;
        Set(varToastTipo; "OK");;
        Set(varToastOn; false);;
        Set(varToastOn; true)
    )
)
```

# Lower(Login) no LookUp de duplicata e so no Desktop, lista pequena. Sem problema.
# Duplicata no editar ignora o proprio ID.
