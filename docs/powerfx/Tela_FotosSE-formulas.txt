                        UpdateIf(

                            CodSE = linhaAtual.CodSE
                            && !Sincronizada
                            && If(
                                !IsBlank(linhaAtual.CodAsset);
                                CodAsset = linhaAtual.CodAsset;
                                !IsBlank(linhaAtual.CodTemp)
                                && CodTemp = linhaAtual.CodTemp
                            );
                            {
                                Sincronizada: true;
                                IdInventario: gravado.ID;
                                CodAsset: chaveSobra
                            }
                        )
                    )
                )
            )
        )
    );;
    ForAll(
        Filter(
            colFotosCampo;
            !Sincronizada
            && CodInventario = gblInventarioSelecionado.CodInventario
            && CodSE in Filter(colSubestacoes; Status = "Finalizada").CodSE
        );
        With(
            {
                fotoAtual: ThisRecord;
                fotoJson: JSON(ThisRecord.Foto; JSONFormat.IncludeBinaryData)
            };
            If(
                !IsBlank(
                    Patch(
                        Fotos;
                        Defaults(Fotos);
                        {
                            Title: Coalesce(fotoAtual.TipoFoto; "ITEM") & " | " & Text(fotoAtual.IdFoto);
                            TipoFoto: Coalesce(fotoAtual.TipoFoto; "ITEM");
                            CodInventario: fotoAtual.CodInventario;
                            CodSE: fotoAtual.CodSE;
                            CodAsset: If(
                                Coalesce(fotoAtual.TipoFoto; "ITEM") = "AMBIENTE";
                                "";
                                Coalesce(
                                    fotoAtual.CodAsset;
                                    If(
                                        !IsBlank(fotoAtual.CodTemp);
                                        "SOBRA-" & Text(fotoAtual.CodTemp);
                                        ""
                                    )
                                )
                            );
                            CodTemp: Coalesce(fotoAtual.CodTemp; "");
                            DataCaptura: fotoAtual.DataCaptura;
                            TiradaPor: gblUsuario.Nome;
                            FotoCampo: {
                                FileName: Text(fotoAtual.IdFoto) & ".jpg";
                                Value: fotoAtual.Foto;
                                Full: fotoJson;
                                Large: fotoJson;
                                Medium: fotoJson;
                                Small: fotoJson
                            }
                        }
                    )
                );
                UpdateIf(
                    colFotosCampo;
                    IdFoto = fotoAtual.IdFoto;
                    {Sincronizada: true}
                )
            )
        )
    );;
    SaveData(colCampoOffline; "cache_CampoOffline");;
    IfError(SaveData(colFotosCampo; "cache_FotosCampo"); Blank());;
    // Offline total 08.11: Local (TI/LI/Arranjo/Coord + gás) sobe só aqui.
    ForAll(
        Filter(
            colSubestacoes;
            CodInventario = gblInventarioSelecionado.CodInventario
            && Status = "Finalizada"
        ) As seFin;
        IfError(
            Patch(
                Subestacoes;
                If(
                    Coalesce(seFin.ID; 0) > 0;
                    LookUp(Subestacoes; ID = seFin.ID);
                    LookUp(
                        Subestacoes;
                        CodSE = seFin.CodSE
                        && CodInventario = seFin.CodInventario
                    )
                );
                {
                    Status: "Finalizada";
                    CoordX: seFin.CoordX;
                    CoordY: seFin.CoordY;
                    TI: Coalesce(seFin.TI; 0);
                    LocalInstalacao: Coalesce(seFin.LocalInstalacao; 0);
                    ArranjoFisico: Coalesce(seFin.ArranjoFisico; 0);
                    NomeSE: seFin.NomeSE;
                    Municipio: seFin.Municipio;
                    Segmento: seFin.Segmento;
                    Tramo: seFin.Tramo
                }
            );
            Blank()
        )
    );;
    IfError(
        flxPastaFotosVisita.Run(gblInventarioSelecionado.CodInventario);
        Blank()
    );;
    With(
        {
            cod: gblInventarioSelecionado.CodInventario;
            todasFinal: CountRows(
                Filter(
                    colSubestacoes;
                    CodInventario = gblInventarioSelecionado.CodInventario
                    && Status <> "Finalizada"
                )
            ) = 0
        };
        Patch(
            'Campanhas ';
            LookUp('Campanhas '; CodInventario = cod);
            {Status: If(todasFinal; "Concluído"; "Em andamento")}
        );;
        UpdateIf(
            colCampanhas;
            CodInventario = cod;
            {Status: If(todasFinal; "Concluído"; "Em andamento")}
        );;
        IfError(SaveData(colCampanhas; "cache_Campanhas"); Blank())
    );;
    Notify("Alterações enviadas com sucesso!"; NotificationType.Success)
);;

Set(varCarregando; false)
