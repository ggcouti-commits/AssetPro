## Galeria — Items (colSEsCampanha + contagens em memoria)

# colSEsCampanha / colInventarioBase / colFotosCampanha carregadas no OnVisible.
# AddColumns roda em memoria (poucas SEs por campanha). Busca por texto fica local.

### galSubestacoesHome.Items

SortByColumns(
    Filter(
        AddColumns(
            AddColumns(
                AddColumns(
                    Filter(
                        colSEsCampanha;
                        Switch(
                            gblFiltroStatusSE;
                            "Todas"; true;
                            "Pendente"; Status = "Pendente";
                            "Em campo"; Status = "Em campo";
                            "Finalizada"; Status = "Finalizada";
                            false
                        )
                    );
                    QtdRevisada;
                    With(
                        {se: CodSE};
                        CountRows(
                            Filter(
                                colInventarioBase;
                                CodSE = se
                                && !IsBlank(Coalesce(ValidadoPor; ""))
                            )
                        )
                    )
                );
                QtdTotal;
                With(
                    {se: CodSE};
                    With(
                        {
                            base: CountRows(
                                Filter(
                                    colInventarioBase;
                                    CodSE = se
                                    && Coalesce(Origem; "") <> "CAMPO"
                                    && Coalesce(Sobra; "") <> "SOBRA_FISICA"
                                    && !StartsWith(Coalesce(Text(CodAsset); ""); "SOBRA-")
                                )
                            );
                            sobras: CountRows(
                                Filter(
                                    colInventarioBase;
                                    CodSE = se
                                    && (
                                        Coalesce(Sobra; "") = "SOBRA_FISICA"
                                        || StartsWith(Coalesce(Text(CodAsset); ""); "SOBRA-")
                                    )
                                )
                            )
                        };
                        base + sobras
                    )
                )
            );
            QtdFotos;
            With(
                {se: CodSE};
                CountRows(
                    Filter(
                        colFotosCampanha;
                        CodSE = se
                    )
                )
            )
        );
        IsBlank(InputBuscaSE.Text)
        || InputBuscaSE.Text in CodSE
        || InputBuscaSE.Text in NomeSE
        || InputBuscaSE.Text in Municipio
    );
    "NomeSE";
    SortOrder.Ascending
)
