    colItensSEConsulta;
    ForAll(
        Filter(
            colInventarioBase;
            CodInventario = gblInventarioSelecionado.CodInventario
            && CodSE = gblSEselecionada.CodSE
        ) As src;
        Patch(
            src;
            {
                CM: src.CM;
                CodAsset: src.CodAsset

        )
    )
);;

Clear(colFotosSEConsulta);;
Collect(
    colFotosSEConsulta;
    ForAll(
        Filter(
            Fotos;
            CodInventario = gblInventarioSelecionado.CodInventario

        Patch(
            f;

                CodAsset: f.CodAsset;

            }
        )

);;

## Galeria — galItensConsulta.Items

SortByColumns(
    Filter(
        AddColumns(
            Filter(
                colItensSEConsulta;
                (
                    IsBlank(InputBuscaItem.Text)
                    || InputBuscaItem.Text in Coalesce(DescricaoBem; "")
                    || InputBuscaItem.Text in Text(Coalesce(CodTUC; ""))
                    || InputBuscaItem.Text in Text(Coalesce(CodAsset; ""))
                    || InputBuscaItem.Text in Coalesce(Imobilizado; "")
                    || InputBuscaItem.Text in Coalesce(IdUC; "")
                )
            ) As itemConsulta;
            StatusConsulta;
            Switch(
                true;
                Coalesce(itemConsulta.Sobra; "") = "SOBRA_FISICA"; "Sobra física";
                Coalesce(itemConsulta.Sobra; "") = "SOBRA_CONTABIL"; "Sobra contábil";
                Coalesce(itemConsulta.Status; "") = "NÃO_ELEGIVEL"; "Sobra contábil";
                Coalesce(itemConsulta.Status; "") = "ADERENTE"; "Confirmado";
                "Pendente"
            );
            QtdFotosItem;
            With(
                {
                    chave: Text(
                        Coalesce(
                            itemConsulta.CodAsset;
                            LookUp(colInventarioBase; ID = itemConsulta.ID; CodAsset);
                            ""
                        )
                    )
                };
                CountRows(
                    Filter(
                        colFotosSEConsulta;
                        Coalesce(
                            TipoFoto;
                            If(StartsWith(Coalesce(Title; ""); "AMBIENTE"); "AMBIENTE"; "ITEM")
                        ) <> "AMBIENTE"
                        && !IsBlank(chave)
                        && Text(Coalesce(CodAsset; "")) = chave
                    )
                )
            )
        );
        Switch(
            gblAbaItensConsulta;
            "BASE";
            Coalesce(Origem; "") <> "CAMPO"
            && Coalesce(Sobra; "") <> "SOBRA_FISICA"
            && !StartsWith(Coalesce(Text(CodAsset); ""); "SOBRA-");
            "SOBRAS";
            Coalesce(Origem; "") = "CAMPO"
            || Coalesce(Sobra; "") = "SOBRA_FISICA"
            || StartsWith(Coalesce(Text(CodAsset); ""); "SOBRA-");
            false
        )
    );
    "IdItem";
    SortOrder.Ascending
)
