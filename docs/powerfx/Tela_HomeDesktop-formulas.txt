## cPainelCampanhas / galCampanhasHome

# Mini-contagens: colSubestacoes (ja filtrada por campanhas ativas no OnStart).
# Revisada = Status Em campo OU Finalizada.

### galCampanhasHome.Items

SortByColumns(
    Filter(
        colCampanhas;
        Status <> "Importando"
        && Switch(
            gblFiltroStatus;
            "Todos"; true;
            "Em andamento"; Status = "Em andamento";
            "Pendente"; Status = "Pendente";
            "Finalizada";
                Status = "Concluído" || Status = "Concluída" || Status = "Concluídas" || Status = "Finalizada";
            "Em campo";
                With(
                    {cod: CodInventario};
                    CountRows(
                        Filter(
                            colSubestacoes;
                            CodInventario = cod && Status = "Em campo"
                        )
                    ) > 0
                );
            false
        )
    );
    "DataCriacao";
    SortOrder.Descending
)

### lblCampMeta.Text

# Subestacao = EhRede <> "Sim". Rede = EhRede = "Sim".
# Numero do mini: esquerda subestacao, direita rede.

With(
    {
        ses: CountRows(
            Filter(
                colSubestacoes;
                CodInventario = ThisItem.CodInventario
                && EhRede <> "Sim"
            )
        );
        redes: CountRows(
            Filter(
                colSubestacoes;
                CodInventario = ThisItem.CodInventario
                && EhRede = "Sim"
            )
        )
    };
    Text(ThisItem.DataCriacao; "mmmm yyyy")
    & " · "
    & ses
    & If(ses = 1; " instalação"; " instalações")
    & " · "
    & redes
    & If(redes = 1; " rede"; " redes")
)

### icoAbrirCampanha.OnSelect

If(
    ThisItem.CodInventario = varLockCod && (ArquivoOcupado || varUltimaOp = "EXCLUINDO");
    With(
        {_: true};
        Set(
            varToastMsg;
            If(
                varUltimaOp = "EXCLUINDO" || varOpArquivo = "EXCLUINDO";
                "Este projeto está sendo excluído.";
                "Este projeto ainda está importando."
            )
        );;
        Set(varToastTipo; "AVISO");;
        Set(varToastOn; false);;
        Set(varToastOn; true)
    );
    Set(varCarregando; true);;
    Set(gblInventarioSelecionado; ThisItem);;
    Set(varCodCampanha; ThisItem.CodInventario);;
    ClearCollect(
        colSEsCampanha;
        Filter(
            colSubestacoes;
            CodInventario = varCodCampanha
        )
    );;
    Clear(colInventarioBase);;
    ForAll(
        colSEsCampanha As se;
        Collect(
            colInventarioBase;
            Filter(
                Inventario;
                CodInventario = varCodCampanha
                && CodSE = se.CodSE
            )
        )
    );;
    ClearCollect(
        colFotosCampanha;
        Filter(
            Fotos;
            CodInventario = varCodCampanha
        )
    );;
    Navigate(Tela_Subestacoes; ScreenTransition.None);;
    Set(varCarregando; false)
)

### lblCampRevisadas.Text

With(
    {
        ses: CountRows(
            Filter(
                colSubestacoes;
                CodInventario = ThisItem.CodInventario
                && EhRede <> "Sim"
            )
        );
        redes: CountRows(
            Filter(
                colSubestacoes;
                CodInventario = ThisItem.CodInventario
                && EhRede = "Sim"
            )
        );
        revSes: CountRows(
            Filter(
                colSubestacoes;
                CodInventario = ThisItem.CodInventario
                && EhRede <> "Sim"
                && (Status = "Em campo" || Status = "Finalizada")
            )
        );
        revRedes: CountRows(
            Filter(
                colSubestacoes;
                CodInventario = ThisItem.CodInventario
                && EhRede = "Sim"
                && (Status = "Em campo" || Status = "Finalizada")
            )
        )
    };
    revSes & " de " & ses & If(ses = 1; " instalação revisada"; " instalações revisadas")
    & " · "
    & revRedes & " de " & redes & If(redes = 1; " rede revisada"; " redes revisadas")
)
