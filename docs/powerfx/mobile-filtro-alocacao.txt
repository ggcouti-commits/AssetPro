## 1. Tela_Login → btnEntrar → OnSelect

# Se o botão ainda se chama Button3, é o Entrar. Apaga o que estiver lá. Cola:

```
Set(varLoginErro; Blank());;
If(
    !Connection.Connected;
    Set(varLoginErro; "Sem internet. O primeiro acesso precisa de rede.");
    With(
        {
            u: LookUp(
                Inventariantes;
                Login = Trim(inpLogin.Text) && Ativo = "Sim"
            )
        };
        If(
            IsBlank(u) || u.Senha <> inpSenha.Text;
            Set(varLoginErro; "Login ou senha inválidos.");
            With(
                {
                    sessao:
                        {
                            Login: u.Login;
                            Perfil: u.Perfil;
                            Nome: Coalesce(u.UsadoPor; u.Nome);
                            UsadoPor: u.UsadoPor;
                            CodUsuario: u.CodUsuario;
                            SessaoVersao: Coalesce(u.SessaoVersao; 0)
                        }
                };
                Set(gblUsuario; sessao);;
                Set(varPerfil; u.Perfil);;
                ClearCollect(colSessao; sessao);;
                IfError(SaveData(colSessao; "cache_Sessao"); Blank());;
                ClearCollect(
                    colAlocacoes;
                    Filter(
                        Alocacoes;
                        Login = u.Login && Ativo = "Sim"
                    )
                );;
                IfError(SaveData(colAlocacoes; "cache_Alocacoes"); Blank());;
                Reset(inpSenha);;
                Navigate(Tela_SelecionarInventario; ScreenTransition.None)
            )
        )
    )
)
```

## 4. Tela_SelecionarSE → Gallery4 → Items

```
SortByColumns(
    Filter(
        AddColumns(
            AddColumns(
                AddColumns(
                    Filter(
                        colSubestacoes;
                        CodInventario = gblInventarioSelecionado.CodInventario
                        && (
                            varPerfil = "ESCRITORIO"
                            || CodSE in Filter(

                        {codSEAtual: CodSE};
                        CountRows(
                            Filter(
                                colCampoOffline;
                                CodSE = codSEAtual
                                && (
                                    TipoRegistro = "CONFIRMACAO"
                                    || TipoRegistro = "NAO_ENCONTRADO"
                                )
                            )
                        )
                    )
                );
                FinalizadaCalc;
                With(
                    {codSEAtual: CodSE};
                    !IsBlank(LookUp(colSEsFinalizadas; CodSE = codSEAtual))
                )
            );
            PendenteEnvioCalc;
            With(
                {codSEAtual: CodSE};
                CountRows(
                    Filter(
                        colCampoOffline;
                        CodSE = codSEAtual && !Sincronizada
                    )
                ) > 0
            )
        );
        (
            gblFiltroStatusSE = "Todas"
            || (
                gblFiltroStatusSE = "Pendente"
                && QtdRevisadaCalc = 0
                && Coalesce(Status; "Pendente") <> "Em campo"
                && Coalesce(Status; "Pendente") <> "Finalizada"
            )
            || (
                gblFiltroStatusSE = "Em campo"
                && (
                    Status = "Em campo"
                    || (Coalesce(Status; "") = "Finalizada" && PendenteEnvioCalc)
                    || (
                        QtdRevisadaCalc > 0
                        && Coalesce(Status; "Pendente") <> "Finalizada"
                    )
                )
            )
            || (
                gblFiltroStatusSE = "Finalizadas"
                && Coalesce(Status; "") = "Finalizada"
                && !PendenteEnvioCalc
            )
        )
        && (
            IsBlank(InputBuscaSE.Text)
            || InputBuscaSE.Text in CodSE
            || InputBuscaSE.Text in NomeSE
        )
    );
    "CodSE";
    SortOrder.Ascending
)
```

## 8. tmrSessao → OnTimerEnd  (e as cópias tmrSessao_1 nas outras telas)

Apaga o que estiver lá. Cola:

```
// Só desloga se o SP devolver o slot de verdade e (inativo ou versão diferente).
// LookUp em branco / erro = mantém logado (não confundir com sinal fraco).
If(
    Connection.Connected && !IsBlank(gblUsuario.Login);
    If(
        IfError(
            With(
                {
                    u: LookUp(
                        Inventariantes;
                        Login = gblUsuario.Login
                    )
                };
                !IsBlank(u.Login)
                    && (
                        u.Ativo <> "Sim"
                        || Coalesce(u.SessaoVersao; 0) <> Coalesce(gblUsuario.SessaoVersao; 0)
                    )
            );
            false
        );
        With(
            {_: true};
            Set(gblUsuario; Blank());;
            Set(varPerfil; Blank());;
            Clear(colSessao);;
            IfError(ClearData("cache_Sessao"); Blank());;
            Clear(colAlocacoes);;
            IfError(ClearData("cache_Alocacoes"); Blank());;
            Reset(inpLogin);;
            Reset(inpSenha);;
            Set(varLoginErro; "Slot alterado. Entra de novo.");
            Navigate(Tela_Login; ScreenTransition.None)
        )
    )
)
```
